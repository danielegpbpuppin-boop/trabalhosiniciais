<script>
    // Configuração inicial e persistência
    let state = JSON.parse(localStorage.getItem('financas_data')) || {
        name: "",
        income: 0,
        expenses: 0,
        step: 'GET_NAME',
        history: [] // Guardamos o histórico do chat
    };

    const chatWindow = document.getElementById('chat-window');
    const mainInput = document.getElementById('main-input');
    const balanceDisplay = document.getElementById('balance-display');
    const userGreeting = document.getElementById('user-greeting');
    const summaryInfo = document.getElementById('summary-info');

    function saveData() {
        localStorage.setItem('financas_data', JSON.stringify(state));
        updateUI();
    }

    function updateUI() {
        const balance = state.income - state.expenses;
        balanceDisplay.innerText = `R$ ${balance.toLocaleString('pt-BR', {minimumFractionDigits: 2})}`;
        userGreeting.innerText = state.name ? `Olá, ${state.name}!` : "Bem-vindo!";
        summaryInfo.innerText = `Renda: R$ ${state.income} | Gastos: R$ ${state.expenses}`;
    }

    function addMessage(text, side, save = true) {
        const div = document.createElement('div');
        div.className = `msg ${side}`;
        div.innerHTML = text;
        chatWindow.appendChild(div);
        chatWindow.scrollTop = chatWindow.scrollHeight;
        
        if (save) {
            state.history.push({text, side});
            saveData();
        }
    }

    // Carregar histórico ao abrir
    function loadHistory() {
        if (state.history.length === 0) {
            addMessage("Olá! Sou o seu assistente. Para começarmos, <b>qual é o seu nome?</b>", 'bot', true);
        } else {
            state.history.forEach(m => addMessage(m.text, m.side, false));
        }
        updateUI();
    }

    function processInput() {
        const val = mainInput.value.trim();
        if (!val) return;

        addMessage(val, 'user');
        mainInput.value = "";

        setTimeout(() => {
            if (state.step === 'GET_NAME') {
                state.name = val;
                state.step = 'GET_INCOME';
                addMessage(`Prazer, ${val}! Agora, qual é o valor da sua <b>renda mensal total</b>?`, 'bot');
            } 
            else if (state.step === 'GET_INCOME') {
                const num = parseFloat(val.replace(',', '.'));
                if (isNaN(num)) {
                    addMessage("Por favor, digite apenas o número (ex: 2500).", 'bot');
                } else {
                    state.income = num;
                    state.step = 'CHAT';
                    addMessage("Tudo pronto! Podes dizer algo como: <i>'Gastei 30 com café'</i> ou <i>'Quanto ainda tenho?'</i>", 'bot');
                }
            } 
            else {
                handleFinanceLogic(val);
            }
            saveData();
        }, 600);
    }

    function handleFinanceLogic(text) {
        const t = text.toLowerCase();
        const numbers = text.match(/\d+([.,]\d+)?/g);

        if (numbers) {
            const value = parseFloat(numbers[0].replace(',', '.'));
            state.expenses += value;
            addMessage(`Anotei! Gasto de <b>R$ ${value.toFixed(2)}</b> registado.`, 'bot');
            
            if (state.expenses > state.income) {
                addMessage("⚠️ <b>Atenção:</b> Os teus gastos ultrapassaram a tua renda!", 'bot');
            }
        } else if (t.includes("quanto") || t.includes("saldo")) {
            const rest = state.income - state.expenses;
            addMessage(`Ainda tens <b>R$ ${rest.toFixed(2)}</b> para gastar este mês.`, 'bot');
        } else {
            addMessage("Não entendi o valor. Tenta dizer 'Gastei 50'.", 'bot');
        }
    }

    function resetarDados() {
        if(confirm("Desejas apagar todos os dados e começar de novo?")) {
            localStorage.removeItem('financas_data');
            location.reload();
        }
    }

    function handleKey(e) { if (e.key === 'Enter') processInput(); }

    window.onload = loadHistory;
</script>
